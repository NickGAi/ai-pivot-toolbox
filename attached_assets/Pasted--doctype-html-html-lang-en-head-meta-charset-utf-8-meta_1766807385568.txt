  <script>
    // -----------------------------
    // AGENCY ROI CALCULATOR
    // -----------------------------
    const tools = [
      "ClickUp","Asana","Monday.com","Trello",
      "HubSpot","Pipedrive","Salesforce",
      "Slack","Google Workspace","Microsoft 365",
      "Notion","Confluence",
      "Zapier","Make.com",
      "Harvest","Toggl","Clockify",
      "Xero","QuickBooks",
      "Figma","Miro"
    ];

    const toolList = document.getElementById("toolList");
    const selected = new Set(["ClickUp","Slack","HubSpot","Notion"]); // defaults

    function currency(n){
      return n.toLocaleString(undefined, { style:"currency", currency:"USD", maximumFractionDigits:0 });
    }

    function renderTools(){
      toolList.innerHTML = "";
      tools.forEach(name=>{
        const id = "t_" + name.replace(/\W/g,"_");
        const row = document.createElement("label");
        row.className = "tool";
        row.setAttribute("for", id);

        const cb = document.createElement("input");
        cb.type = "checkbox";
        cb.id = id;
        cb.checked = selected.has(name);
        cb.addEventListener("change", ()=>{
          if(cb.checked) selected.add(name); else selected.delete(name);
          calc();
        });

        const txt = document.createElement("div");
        txt.textContent = name;

        row.appendChild(cb);
        row.appendChild(txt);
        toolList.appendChild(row);
      });
    }

    const elAdmin = document.getElementById("adminHours");
    const elMeet  = document.getElementById("meetHours");
    const elTeam  = document.getElementById("team");
    const elRate  = document.getElementById("rate");
    const elToolCost = document.getElementById("toolCost");
    const elRework = document.getElementById("reworkPct");

    const elWaste = document.getElementById("waste");
    const elFormula = document.getElementById("formula");

    function clampNum(v, min, fallback){
      const n = Number(v);
      if(Number.isFinite(n) && n >= min) return n;
      return fallback;
    }

    function calc(){
      const admin = clampNum(elAdmin.value, 0, 6);
      const meet  = clampNum(elMeet.value, 0, 3);
      const team  = clampNum(elTeam.value, 1, 8);
      const rate  = clampNum(elRate.value, 0, 65);
      const toolCost = clampNum(elToolCost.value, 0, 25);
      const reworkPct = clampNum(elRework.value, 0, 8);

      const toolCount = selected.size;

      // annual SaaS
      const annualSaaS = toolCount * toolCost * team * 12;

      // annual coordination time
      const weeklyCoordHours = (admin + meet) * team;
      const annualCoord = weeklyCoordHours * rate * 52;

      // rework estimate as % of total hours (admin+meet considered part of total time here)
      const weeklyTotalHours = (admin + meet) * team;
      const weeklyReworkHours = weeklyTotalHours * (reworkPct / 100);
      const annualRework = weeklyReworkHours * rate * 52;

      const total = annualSaaS + annualCoord + annualRework;

      elWaste.textContent = currency(total);

      elFormula.textContent =
        `SaaS: ${currency(annualSaaS)} (${toolCount} tools × ${currency(toolCost)}/user/mo × ${team} × 12) + ` +
        `Coordination: ${currency(annualCoord)} ((${admin}+${meet}) hrs/wk × ${team} × ${currency(rate)}/hr × 52) + ` +
        `Rework: ${currency(annualRework)} (${reworkPct}% of coordination time)`;
    }

    [elAdmin, elMeet, elTeam, elRate, elToolCost, elRework].forEach(el => el.addEventListener("input", calc));
    document.getElementById("yr").textContent = new Date().getFullYear();

    renderTools();
    calc();
  </script>
